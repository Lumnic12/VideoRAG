#### Task 1.2: SSIM Frame Extractor (The Critical Algorithm)

```python
# backend/services/frame_extractor.py
"""
SSIM-based keyframe extraction.
Drops perceptually identical frames to reduce thousands of frames 
to ~10-15 semantic keyframes.
"""
import cv2
import numpy as np
from skimage.metrics import structural_similarity as ssim
from pathlib import Path
from dataclasses import dataclass
import structlog

logger = structlog.get_logger()


@dataclass
class Keyframe:
    index: int
    timestamp_sec: float
    frame: np.ndarray          # BGR numpy array
    ssim_delta: float          # How different from previous keyframe


def extract_keyframes(
    video_path: str,
    threshold: float = 0.95,
    min_interval_sec: float = 2.0,
    max_keyframes: int = 30,
    resize_width: int = 640,
) -> list[Keyframe]:
    """
    Core SSIM loop. Performance-critical path.
    
    Args:
        video_path: Path to .mp4 file
        threshold: SSIM score above which frames are considered identical
        min_interval_sec: Minimum gap between keyframes to avoid duplicates
        max_keyframes: Hard cap to prevent memory blowup
        resize_width: Downscale for faster SSIM computation
        
    Returns:
        List of Keyframe objects with timestamps
    """
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened():
        raise ValueError(f"Cannot open video: {video_path}")

    fps = cap.get(cv2.CAP_PROP_FPS)
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    duration_sec = total_frames / fps
    
    logger.info(
        "extraction_started",
        fps=fps,
        total_frames=total_frames,
        duration_sec=round(duration_sec, 1),
    )

    # Performance optimization: skip frames instead of reading every one
    # For a 30fps video, reading every 15th frame (2x per second) is sufficient
    sample_interval = max(1, int(fps // 2))
    
    keyframes: list[Keyframe] = []
    prev_gray = None
    frame_idx = 0

    while True:
        cap.set(cv2.CAP_PROP_POS_FRAMES, frame_idx)
        ret, frame = cap.read()
        if not ret:
            break

        # Downscale for SSIM speed — we keep the original resolution frame
        h, w = frame.shape[:2]
        scale = resize_width / w
        small = cv2.resize(frame, (resize_width, int(h * scale)))
        gray = cv2.cvtColor(small, cv2.COLOR_BGR2GRAY)

        if prev_gray is None:
            # First frame is always a keyframe
            timestamp = frame_idx / fps
            keyframes.append(Keyframe(
                index=len(keyframes),
                timestamp_sec=round(timestamp, 2),
                frame=frame.copy(),
                ssim_delta=0.0,
            ))
            prev_gray = gray
            frame_idx += sample_interval
            continue

        score = ssim(prev_gray, gray)
        timestamp = frame_idx / fps

        # Check minimum interval constraint
        time_since_last = timestamp - keyframes[-1].timestamp_sec

        if score < threshold and time_since_last >= min_interval_sec:
            keyframes.append(Keyframe(
                index=len(keyframes),
                timestamp_sec=round(timestamp, 2),
                frame=frame.copy(),
                ssim_delta=round(1.0 - score, 4),
            ))
            prev_gray = gray
            logger.debug(
                "keyframe_captured",
                index=len(keyframes) - 1,
                timestamp=round(timestamp, 2),
                ssim=round(score, 4),
            )

            if len(keyframes) >= max_keyframes:
                logger.warning("max_keyframes_reached", cap=max_keyframes)
                break

        frame_idx += sample_interval

    cap.release()
    logger.info("extraction_complete", keyframes_found=len(keyframes))
    return keyframes


def save_keyframes(keyframes: list[Keyframe], output_dir: str) -> list[str]:
    """Save keyframe images to disk. Returns list of file paths."""
    out = Path(output_dir)
    out.mkdir(parents=True, exist_ok=True)
    paths = []
    for kf in keyframes:
        filename = f"kf_{kf.index:03d}_{kf.timestamp_sec:.1f}s.jpg"
        path = str(out / filename)
        cv2.imwrite(path, kf.frame, [cv2.IMWRITE_JPEG_QUALITY, 90])
        paths.append(path)
    return paths
```

#### Task 1.3: VLM Service (Jina-VLM via Hugging Face Inference)

```python
# backend/services/vlm_service.py
"""
Calls Jina-VLM through Hugging Face Inference API.
Processes keyframe images → returns structured text extraction.
"""
import httpx
import base64
import asyncio
from pathlib import Path
from dataclasses import dataclass
from core.config import settings
import structlog

logger = structlog.get_logger()

HF_INFERENCE_URL = "https://router.huggingface.co/hf-inference/models/jinaai/jina-vlm-v1"

@dataclass
class SlideAnalysis:
    keyframe_index: int
    timestamp_sec: float
    extracted_text: str
    slide_title: str
    code_snippets: list[str]
    key_concepts: list[str]


def _encode_image(image_path: str) -> str:
    with open(image_path, "rb") as f:
        return base64.b64encode(f.read()).decode("utf-8")


async def analyze_keyframe(
    image_path: str,
    keyframe_index: int,
    timestamp_sec: float,
) -> SlideAnalysis:
    """Send a single keyframe to Jina-VLM for analysis."""
    
    b64 = _encode_image(image_path)
    
    prompt = """Analyze this lecture slide image. Return a JSON object with:
{
  "slide_title": "the title or heading visible",
  "extracted_text": "all readable text on the slide",
  "code_snippets": ["any code blocks found"],
  "key_concepts": ["3-5 key technical concepts mentioned"]
}
Only return valid JSON. No markdown fences."""

    payload = {
        "inputs": [
            {
                "type": "image_url",
                "image_url": {"url": f"data:image/jpeg;base64,{b64}"},
            },
            {"type": "text", "text": prompt},
        ],
        "parameters": {"max_new_tokens": 512, "temperature": 0.1},
    }

    headers = {"Authorization": f"Bearer {settings.hf_api_token}"}

    async with httpx.AsyncClient(timeout=60.0) as client:
        resp = await client.post(HF_INFERENCE_URL, json=payload, headers=headers)
        resp.raise_for_status()
        result = resp.json()

    # Parse the model output — defensive handling
    try:
        import json
        text_output = result[0]["generated_text"] if isinstance(result, list) else result.get("generated_text", "")
        parsed = json.loads(text_output)
    except (json.JSONDecodeError, KeyError, IndexError):
        logger.warning("vlm_parse_failed", raw=str(result)[:200])
        parsed = {
            "slide_title": "",
            "extracted_text": str(result),
            "code_snippets": [],
            "key_concepts": [],
        }

    return SlideAnalysis(
        keyframe_index=keyframe_index,
        timestamp_sec=timestamp_sec,
        extracted_text=parsed.get("extracted_text", ""),
        slide_title=parsed.get("slide_title", ""),
        code_snippets=parsed.get("code_snippets", []),
        key_concepts=parsed.get("key_concepts", []),
    )


async def analyze_all_keyframes(image_paths: list[str], timestamps: list[float]) -> list[SlideAnalysis]:
    """Process all keyframes with bounded concurrency."""
    semaphore = asyncio.Semaphore(3)  # Max 3 concurrent API calls
    
    async def _bounded(path, idx, ts):
        async with semaphore:
            return await analyze_keyframe(path, idx, ts)
    
    tasks = [
        _bounded(path, idx, ts)
        for idx, (path, ts) in enumerate(zip(image_paths, timestamps))
    ]
    return await asyncio.gather(*tasks)
```

```python
# backend/services/audio_service.py
"""
Deepgram integration for audio transcription.
Extracts audio from video, sends to Deepgram, returns timestamped transcript.
"""
import subprocess
import asyncio
from pathlib import Path
from dataclasses import dataclass
from deepgram import DeepgramClient, PrerecordedOptions
from core.config import settings
import structlog

logger = structlog.get_logger()


@dataclass
class TranscriptSegment:
    start: float
    end: float
    text: str


def extract_audio(video_path: str, output_dir: str) -> str:
    """Extract audio track from video using ffmpeg."""
    audio_path = str(Path(output_dir) / "audio.wav")
    cmd = [
        "ffmpeg", "-i", video_path,
        "-vn",                    # no video
        "-acodec", "pcm_s16le",   # WAV format
        "-ar", "16000",           # 16kHz sample rate (optimal for speech)
        "-ac", "1",               # mono
        "-y",                     # overwrite
        audio_path,
    ]
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=120)
    if result.returncode != 0:
        raise RuntimeError(f"ffmpeg failed: {result.stderr[:300]}")
    logger.info("audio_extracted", path=audio_path)
    return audio_path


async def transcribe_audio(audio_path: str) -> list[TranscriptSegment]:
    """Send audio to Deepgram, return timestamped segments."""
    client = DeepgramClient(settings.deepgram_api_key)

    with open(audio_path, "rb") as f:
        audio_data = f.read()

    options = PrerecordedOptions(
        model="nova-2",
        smart_format=True,
        utterances=True,
        punctuate=True,
    )

    response = await asyncio.to_thread(
        client.listen.prerecorded.v("1").transcribe_file,
        {"buffer": audio_data, "mimetype": "audio/wav"},
        options,
    )

    segments = []
    for utterance in response.results.utterances:
        segments.append(TranscriptSegment(
            start=utterance.start,
            end=utterance.end,
            text=utterance.transcript,
        ))

    logger.info("transcription_complete", segments=len(segments))
    return segments
```

#### Task 1.5: Celery Task Orchestrator

```python
# backend/services/task_orchestrator.py
"""
Celery tasks that wire the pipeline together.
Each step updates job status in Redis.
"""
from celery import Celery
from core.config import settings
import json
import redis
import asyncio

celery_app = Celery("svs", broker=settings.redis_url, backend=settings.redis_url)
redis_client = redis.from_url(settings.redis_url)


def update_job(job_id: str, status: str, progress: int, data: dict = None):
    """Write job state to Redis for frontend polling."""
    state = {"status": status, "progress": progress}
    if data:
        state["result"] = data
    redis_client.set(f"job:{job_id}", json.dumps(state), ex=3600)


@celery_app.task(name="process_video")
def process_video_task(job_id: str, video_path: str):
    """Main pipeline orchestration — runs in Celery worker process."""
    from services.frame_extractor import extract_keyframes, save_keyframes
    from services.vlm_service import analyze_all_keyframes
    from services.audio_service import extract_audio, transcribe_audio
    from services.rag_service import index_content

    try:
        # Step 1: Extract keyframes (SSIM)
        update_job(job_id, "processing", 10)
        keyframes = extract_keyframes(video_path, threshold=settings.ssim_threshold)
        
        job_keyframe_dir = f"{settings.keyframe_dir}/{job_id}"
        image_paths = save_keyframes(keyframes, job_keyframe_dir)
        timestamps = [kf.timestamp_sec for kf in keyframes]
        update_job(job_id, "processing", 30)

        # Step 2: Extract + transcribe audio (parallel with VLM)
        audio_path = extract_audio(video_path, job_keyframe_dir)
        
        loop = asyncio.new_event_loop()
        
        # Step 3: Run VLM + transcription concurrently
        async def parallel_processing():
            vlm_task = analyze_all_keyframes(image_paths, timestamps)
            transcript_task = transcribe_audio(audio_path)
            return await asyncio.gather(vlm_task, transcript_task)
        
        update_job(job_id, "processing", 50)
        analyses, transcript = loop.run_until_complete(parallel_processing())
        loop.close()
        update_job(job_id, "processing", 80)

        # Step 4: Build result
        result = {
            "keyframes": [
                {
                    "index": a.keyframe_index,
                    "timestamp": a.timestamp_sec,
                    "image_url": f"/static/keyframes/{job_id}/{Path(p).name}",
                    "extracted_text": a.extracted_text,
                    "slide_title": a.slide_title,
                    "code_snippets": a.code_snippets,
                    "key_concepts": a.key_concepts,
                }
                for a, p in zip(analyses, image_paths)
            ],
            "transcript": [
                {"start": s.start, "end": s.end, "text": s.text}
                for s in transcript
            ],
        }

        # Step 5: Index into RAG
        index_content(job_id, result)
        
        update_job(job_id, "complete", 100, result)

    except Exception as e:
        update_job(job_id, "failed", 0, {"error": str(e)})
        raise
```

# backend/api/routes/video.py
from fastapi import APIRouter, UploadFile, File, HTTPException
from fastapi.responses import JSONResponse
import uuid
import aiofiles
from pathlib import Path
from core.config import settings
from services.task_orchestrator import process_video_task, redis_client
import json

router = APIRouter(prefix="/api/v1", tags=["video"])


@router.post("/process")
async def process_video(video: UploadFile = File(...)):
    # Validate
    if not video.filename.endswith((".mp4", ".mov", ".avi", ".mkv")):
        raise HTTPException(400, "Unsupported video format")

    # Save uploaded file
    job_id = str(uuid.uuid4())
    save_path = Path(settings.upload_dir) / f"{job_id}_{video.filename}"
    
    async with aiofiles.open(save_path, "wb") as f:
        content = await video.read()
        if len(content) > settings.max_upload_size_mb * 1024 * 1024:
            raise HTTPException(413, "File too large")
        await f.write(content)

    # Dispatch to Celery
    process_video_task.delay(job_id, str(save_path))
    
    return {"job_id": job_id, "status": "queued"}


@router.get("/jobs/{job_id}")
async def get_job_status(job_id: str):
    raw = redis_client.get(f"job:{job_id}")
    if not raw:
        raise HTTPException(404, "Job not found")
    return json.loads(raw)

// frontend/src/hooks/useJobPolling.ts
import { useEffect, useRef } from 'react';
import { useJobStore } from '@/store/useJobStore';

export function useJobPolling() {
  const { jobId, status, setStatus, setProgress, setResult } = useJobStore();
  const intervalRef = useRef<NodeJS.Timeout>();

  useEffect(() => {
    if (!jobId || status === 'complete' || status === 'failed') return;

    intervalRef.current = setInterval(async () => {
      const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL}/api/v1/jobs/${jobId}`);
      const data = await res.json();
      
      setProgress(data.progress);
      setStatus(data.status);
      
      if (data.status === 'complete') {
        setResult(data.result);
        clearInterval(intervalRef.current);
      }
      if (data.status === 'failed') {
        clearInterval(intervalRef.current);
      }
    }, 1000);  // Poll every second

    return () => clearInterval(intervalRef.current);
  }, [jobId, status]);
}

### PHASE 3 — RAG "Second Brain" (Days 10–12)

```python
# backend/services/rag_service.py
"""
FAISS-based local vector store + OpenAI embeddings.
Stores all processed lecture content for cross-video querying.
"""
import faiss
import numpy as np
import json
from pathlib import Path
from openai import OpenAI
from core.config import settings
import structlog

logger = structlog.get_logger()

client = OpenAI(api_key=settings.openai_api_key)

# In-memory metadata store (SQLite in production)
_metadata_store: dict[int, dict] = {}
_index: faiss.IndexFlatIP | None = None
_counter = 0


def _get_embedding(text: str) -> np.ndarray:
    resp = client.embeddings.create(input=text, model="text-embedding-3-small")
    return np.array(resp.data[0].embedding, dtype="float32")


def _get_or_create_index() -> faiss.IndexFlatIP:
    global _index
    if _index is None:
        _index = faiss.IndexFlatIP(1536)  # text-embedding-3-small dimension
        # Load existing index if available
        idx_path = Path(settings.faiss_index_path) / "index.faiss"
        meta_path = Path(settings.faiss_index_path) / "metadata.json"
        if idx_path.exists():
            _index = faiss.read_index(str(idx_path))
            with open(meta_path) as f:
                global _metadata_store, _counter
                _metadata_store = {int(k): v for k, v in json.load(f).items()}
                _counter = max(_metadata_store.keys(), default=-1) + 1
    return _index


def _save_index():
    idx_path = Path(settings.faiss_index_path) / "index.faiss"
    meta_path = Path(settings.faiss_index_path) / "metadata.json"
    Path(settings.faiss_index_path).mkdir(parents=True, exist_ok=True)
    faiss.write_index(_get_or_create_index(), str(idx_path))
    with open(meta_path, "w") as f:
        json.dump(_metadata_store, f)


def index_content(video_id: str, result: dict):
    """Chunk and embed all content from a processed video."""
    global _counter
    index = _get_or_create_index()
    vectors = []

    # Index keyframe content
    for kf in result["keyframes"]:
        chunk = f"[Slide: {kf['slide_title']}] {kf['extracted_text']}"
        if kf["code_snippets"]:
            chunk += f" Code: {' '.join(kf['code_snippets'])}"

        emb = _get_embedding(chunk)
        _metadata_store[_counter] = {
            "video_id": video_id,
            "type": "keyframe",
            "timestamp": kf["timestamp"],
            "text": chunk,
            "image_url": kf["image_url"],
        }
        vectors.append(emb)
        _counter += 1

    # Index transcript in ~30-second windows
    transcript = result["transcript"]
    window_text = ""
    window_start = 0.0
    for seg in transcript:
        window_text += " " + seg["text"]
        if seg["end"] - window_start >= 30.0:
            emb = _get_embedding(window_text.strip())
            _metadata_store[_counter] = {
                "video_id": video_id,
                "type": "transcript",
                "timestamp": window_start,
                "text": window_text.strip(),
            }
            vectors.append(emb)
            _counter += 1
            window_start = seg["end"]
            window_text = ""

    if vectors:
        mat = np.stack(vectors)
        faiss.normalize_L2(mat)
        index.add(mat)
        _save_index()
    
    logger.info("rag_indexed", video_id=video_id, chunks=len(vectors))


def query(question: str, top_k: int = 5, video_ids: list[str] | None = None) -> list[dict]:
    """Retrieve the most relevant chunks for a question."""
    index = _get_or_create_index()
    if index.ntotal == 0:
        return []

    q_emb = _get_embedding(question).reshape(1, -1)
    faiss.normalize_L2(q_emb)
    scores, indices = index.search(q_emb, top_k * 2)  # over-fetch then filter

    results = []
    for score, idx in zip(scores[0], indices[0]):
        if idx == -1:
            continue
        meta = _metadata_store.get(int(idx))
        if not meta:
            continue
        if video_ids and meta["video_id"] not in video_ids:
            continue
        results.append({**meta, "score": float(score)})
        if len(results) >= top_k:
            break

    return results
```

# backend/api/routes/query.py
from fastapi import APIRouter
from pydantic import BaseModel
from services.rag_service import query
from openai import OpenAI
from core.config import settings

router = APIRouter(prefix="/api/v1", tags=["query"])
client = OpenAI(api_key=settings.openai_api_key)


class QueryRequest(BaseModel):
    question: str
    video_ids: list[str] | None = None


@router.post("/query")
async def ask_second_brain(req: QueryRequest):
    sources = query(req.question, top_k=5, video_ids=req.video_ids)
    
    if not sources:
        return {"answer": "No relevant content found. Process some videos first.", "sources": []}

    context = "\n\n".join([
        f"[{s['type']} @ {s['timestamp']:.1f}s] {s['text']}" 
        for s in sources
    ])

    completion = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": "You are a study assistant. Answer based ONLY on the provided lecture content. Cite timestamps."},
            {"role": "user", "content": f"Context:\n{context}\n\nQuestion: {req.question}"},
        ],
        temperature=0.2,
        max_tokens=500,
    )

    return {
        "answer": completion.choices[0].message.content,
        "sources": [
            {"video_id": s["video_id"], "timestamp": s["timestamp"], "chunk_text": s["text"][:200]}
            for s in sources
        ],
    }

#### Task 4.1: C++ SSIM (only after Python MVP works)

```cpp
// cpp_bindings/ssim_extractor.cpp
#include <pybind11/pybind11.h>
#include <pybind11/numpy.h>
#include <pybind11/stl.h>
#include <opencv2/opencv.hpp>
#include <vector>

namespace py = pybind11;

struct KeyframeResult {
    int index;
    double timestamp_sec;
    double ssim_delta;
};

double compute_ssim(const cv::Mat& img1, const cv::Mat& img2) {
    cv::Mat i1, i2;
    img1.convertTo(i1, CV_64F);
    img2.convertTo(i2, CV_64F);

    double C1 = 6.5025, C2 = 58.5225;  // (0.01*255)^2, (0.03*255)^2

    cv::Mat mu1, mu2;
    cv::GaussianBlur(i1, mu1, cv::Size(11, 11), 1.5);
    cv::GaussianBlur(i2, mu2, cv::Size(11, 11), 1.5);

    cv::Mat mu1_sq = mu1.mul(mu1);
    cv::Mat mu2_sq = mu2.mul(mu2);
    cv::Mat mu1_mu2 = mu1.mul(mu2);

    cv::Mat sigma1_sq, sigma2_sq, sigma12;
    cv::GaussianBlur(i1.mul(i1), sigma1_sq, cv::Size(11, 11), 1.5);
    sigma1_sq -= mu1_sq;
    cv::GaussianBlur(i2.mul(i2), sigma2_sq, cv::Size(11, 11), 1.5);
    sigma2_sq -= mu2_sq;
    cv::GaussianBlur(i1.mul(i2), sigma12, cv::Size(11, 11), 1.5);
    sigma12 -= mu1_mu2;

    cv::Mat ssim_map;
    cv::Mat num = (2 * mu1_mu2 + C1).mul(2 * sigma12 + C2);
    cv::Mat den = (mu1_sq + mu2_sq + C1).mul(sigma1_sq + sigma2_sq + C2);
    cv::divide(num, den, ssim_map);

    return cv::mean(ssim_map)[0];
}

std::vector<KeyframeResult> extract_keyframes_cpp(
    const std::string& video_path,
    double threshold,
    double min_interval,
    int max_keyframes,
    int resize_width
) {
    cv::VideoCapture cap(video_path);
    if (!cap.isOpened()) throw std::runtime_error("Cannot open: " + video_path);

    double fps = cap.get(cv::CAP_PROP_FPS);
    int total = (int)cap.get(cv::CAP_PROP_FRAME_COUNT);
    int sample_interval = std::max(1, (int)(fps / 2));

    std::vector<KeyframeResult> results;
    cv::Mat prev_gray, frame, small, gray;

    for (int idx = 0; idx < total; idx += sample_interval) {
        cap.set(cv::CAP_PROP_POS_FRAMES, idx);
        if (!cap.read(frame)) break;

        double scale = (double)resize_width / frame.cols;
        cv::resize(frame, small, cv::Size(resize_width, (int)(frame.rows * scale)));
        cv::cvtColor(small, gray, cv::COLOR_BGR2GRAY);

        if (prev_gray.empty()) {
            results.push_back({0, idx / fps, 0.0});
            gray.copyTo(prev_gray);
            continue;
        }

        double score = compute_ssim(prev_gray, gray);
        double timestamp = idx / fps;
        double time_since_last = timestamp - results.back().timestamp_sec;

        if (score < threshold && time_since_last >= min_interval) {
            results.push_back({(int)results.size(), timestamp, 1.0 - score});
            gray.copyTo(prev_gray);
            if ((int)results.size() >= max_keyframes) break;
        }
    }
    return results;
}

PYBIND11_MODULE(ssim_cpp, m) {
    py::class_<KeyframeResult>(m, "KeyframeResult")
        .def_readonly("index", &KeyframeResult::index)
        .def_readonly("timestamp_sec", &KeyframeResult::timestamp_sec)
        .def_readonly("ssim_delta", &KeyframeResult::ssim_delta);

    m.def("extract_keyframes", &extract_keyframes_cpp,
        py::arg("video_path"),
        py::arg("threshold") = 0.95,
        py::arg("min_interval") = 2.0,
        py::arg("max_keyframes") = 30,
        py::arg("resize_width") = 640
    );
}
```
